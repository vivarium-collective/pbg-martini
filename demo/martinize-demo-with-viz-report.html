name: martinize-demo-with-viz
description: "Martini coarse-graining demo with a wired Visualization Step. Each update produces a Plotly HTML figure of the running CG summary scalars, written to the 'viz_html' store."

requires:
  processes: [MartinizeStep, MartiniBeadSummaryPlots, RAMEmitter]

parameters:
  from_ff:
    type: string
    default: "charmm"
    description: "Source atomistic force field."
  to_ff:
    type: string
    default: "martini3001"
    description: "Target Martini force field."
  viz_title:
    type: string
    default: "Martini CG mapping summary"
    description: "Title displayed at the top of the Plotly figure."
  pdb_text:
    type: string
    default: |
      ATOM      1  N   ALA A   1       1.000   1.000   1.000  1.00  0.00           N
      ATOM      2  CA  ALA A   1       2.450   1.000   1.000  1.00  0.00           C
      ATOM      3  C   ALA A   1       3.000   2.400   1.000  1.00  0.00           C
      ATOM      4  O   ALA A   1       2.400   3.400   1.000  1.00  0.00           O
      ATOM      5  CB  ALA A   1       3.000   0.200   2.200  1.00  0.00           C
      ATOM      6  N   ALA A   2       4.300   2.400   1.000  1.00  0.00           N
      ATOM      7  CA  ALA A   2       5.000   3.700   1.000  1.00  0.00           C
      ATOM      8  C   ALA A   2       6.500   3.700   1.000  1.00  0.00           C
      ATOM      9  O   ALA A   2       7.100   4.700   1.000  1.00  0.00           O
      ATOM     10  CB  ALA A   2       4.500   4.500   2.200  1.00  0.00           C
      ATOM     11  N   ALA A   3       7.100   2.600   1.000  1.00  0.00           N
      ATOM     12  CA  ALA A   3       8.500   2.500   1.000  1.00  0.00           C
      ATOM     13  C   ALA A   3       9.100   3.800   1.000  1.00  0.00           C
      ATOM     14  O   ALA A   3       8.500   4.800   1.000  1.00  0.00           O
      ATOM     15  CB  ALA A   3       9.000   1.700   2.200  1.00  0.00           C
      END
    description: "PDB-format atomistic structure to coarse-grain."

state:
  martinize:
    _type: step
    address: "local:MartinizeStep"
    config:
      from_ff: "${from_ff}"
      to_ff: "${to_ff}"
      delete_unknown: true
      ignh: false
    inputs:
      pdb_text: ["stores", "pdb_text"]
    outputs:
      cg_beads: ["stores", "cg_beads"]
      cg_positions: ["stores", "cg_positions"]
      cg_bonds: ["stores", "cg_bonds"]
      interactions: ["stores", "interactions"]
      bead_type_counts: ["stores", "bead_type_counts"]
      residue_counts: ["stores", "residue_counts"]
      n_atoms_input: ["stores", "n_atoms_input"]
      n_atoms_full: ["stores", "n_atoms_full"]
      n_beads: ["stores", "n_beads"]
      n_bonds_cg: ["stores", "n_bonds_cg"]
      reduction_ratio: ["stores", "reduction_ratio"]
  stores:
    pdb_text: "${pdb_text}"
    n_atoms_input: 0
    n_atoms_full: 0
    n_beads: 0
    n_bonds_cg: 0
    reduction_ratio: 0.0
    viz_html: ""
  viz:
    _type: step
    address: "local:MartiniBeadSummaryPlots"
    config:
      title: "${viz_title}"
    inputs:
      n_atoms_input: ["stores", "n_atoms_input"]
      n_atoms_full: ["stores", "n_atoms_full"]
      n_beads: ["stores", "n_beads"]
      n_bonds_cg: ["stores", "n_bonds_cg"]
      reduction_ratio: ["stores", "reduction_ratio"]
      time: ["global_time"]
    outputs:
      html: ["stores", "viz_html"]
  emitter:
    _type: step
    address: "local:RAMEmitter"
    config:
      emit:
        n_atoms_input: "integer"
        n_atoms_full: "integer"
        n_beads: "integer"
        n_bonds_cg: "integer"
        reduction_ratio: "float"
    inputs:
      n_atoms_input: ["stores", "n_atoms_input"]
      n_atoms_full: ["stores", "n_atoms_full"]
      n_beads: ["stores", "n_beads"]
      n_bonds_cg: ["stores", "n_bonds_cg"]
      reduction_ratio: ["stores", "reduction_ratio"]
